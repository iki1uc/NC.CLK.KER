2. JS — gleiche Regel

// quetschen.js — Alles quetschen, JS
const SCORE={"·":0,"❄":0.4,"≈":0.65,"✿":1,"✕":0,"▲":0.85};
const f=x=>Math.min(x,40);
function quetschen(layers){
  const v81=layers.map(l=>l.reduce((s,c)=>s+(SCORE[c]??0),0)/81);
  const v9=Array.from({length:9},(_,i)=>v81.filter((_,j)=>j%9===i).reduce((a,b)=>a+b,0)/9);
  // Achtung: gleiche Spalten-Regel wie Python: v81[j*9+i]
  const v9b=Array.from({length:9},(_,i)=>[0,1,2,3,4,5,6,7,8].reduce((s,j)=>s+v81[j*9+i],0)/9);
  const v3=[(v9b[0]+v9b[1]+v9b[2])/3,(v9b[3]+v9b[4]+v9b[5])/3,(v9b[6]+v9b[7]+v9b[8])/3];
  const core=(v3[0]+v3[1]+v3[2])/3;
  return {v81,v9:v9b,v3,core:f(core*40)/40,core_roh:core};
}

Python und JS müssen dieselbe Zahl geben. Sonst ist einer falsch. Hier: beide geben 1.0 bei alle ✿, 0.0 bei alle ·.
